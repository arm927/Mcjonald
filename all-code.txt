import { useEffect } from 'react';

export default function Modal({ title, onClose, children }) {
  useEffect(() => {
    const onKey = (e) => e.key === 'Escape' && onClose();
    document.addEventListener('keydown', onKey);
    document.body.style.overflow = 'hidden';
    return () => {
      document.removeEventListener('keydown', onKey);
      document.body.style.overflow = '';
    };
  }, [onClose]);

  return (
    <div className="overlay" onMouseDown={(e) => e.target === e.currentTarget && onClose()}>
      <div className="modal" role="dialog" aria-modal="true" aria-label={title}>
        <div className="modal-head">
          <h2>{title}</h2>
          <button className="icon-btn" onClick={onClose} aria-label="Close">✕</button>
        </div>
        {children}
      </div>
    </div>
  );
}

import { useEffect } from 'react';

export default function Toast({ toast, onDone }) {
  useEffect(() => {
    if (!toast) return;
    const t = setTimeout(onDone, 2800);
    return () => clearTimeout(t);
  }, [toast, onDone]);

  if (!toast) return null;
  return (
    <div className={`toast toast-${toast.type}`} role="status" aria-live="polite">
      {toast.message}
    </div>
  );
}

    e.preventDefault();
    const err = {};
    if (!form.name.trim()) err.name = 'Name is required';
    if (form.price === '' || Number(form.price) <= 0) err.price = 'Enter a price above 0';
    setErrors(err);
    if (Object.keys(err).length) return;
    onSubmit({ ...form, name: form.name.trim(), description: form.description.trim(), price: Number(form.price) });
  };

  return (
    <form onSubmit={submit} noValidate>
      <label>
        Item name
        <input value={form.name} onChange={(e) => set('name', e.target.value)} placeholder="e.g. Caramel Macchiato" autoFocus />
        {errors.name && <span className="error">{errors.name}</span>}
      </label>
      <div className="row">
        <label>
          Category
          <select value={form.category} onChange={(e) => set('category', e.target.value)}>
            {CATEGORIES.map((c) => <option key={c}>{c}</option>)}
          </select>
        </label>
        <label>
          Price (₱)
          <input type="number" min="0" step="1" value={form.price} onChange={(e) => set('price', e.target.value)} />

        </label>
      </div>

function OrderForm({ initial, menu, onSubmit, onCancel }) {

  const [status, setStatus] = useState(initial?.status || 'Pending');
  const [note, setNote] = useState(initial?.note || '');
  const [lines, setLines] = useState(initial?.items?.length ? initial.items : [{ menuId: '', qty: 1 }]);

const orderTotal = (order, menu) =>
  order.items.reduce((sum, line) => sum + (menu.find((m) => m.id === line.menuId)?.price || 0) * line.qty, 0);

function OrderForm({ initial, menu, onSubmit, onCancel }) {
  const [customer, setCustomer] = useState(initial?.customer || '');
  const [status, setStatus] = useState(initial?.status || 'Pending');
  const [note, setNote] = useState(initial?.note || '');
  const [lines, setLines] = useState(initial?.items?.length ? initial.items : [{ menuId: '', qty: 1 }]);
  const [errors, setErrors] = useState({});

  const setLine = (i, patch) => setLines((ls) => ls.map((l, idx) => (idx === i ? { ...l, ...patch } : l)));
  const removeLine = (i) => setLines((ls) => (ls.length === 1 ? ls : ls.filter((_, idx) => idx !== i)));
  const total = orderTotal({ items: lines }, menu);

  const submit = (e) => {
    e.preventDefault();
    const err = {};
    if (!customer.trim()) err.customer = 'Customer name is required';
    const valid = lines.filter((l) => l.menuId);
    if (!valid.length) err.items = 'Add at least one item';
    setErrors(err);
    if (Object.keys(err).length) return;
    onSubmit({ customer: customer.trim(), status, note: note.trim(), items: valid.map((l) => ({ menuId: l.menuId, qty: Number(l.qty) || 1 })) });
  };

  return (
    <form onSubmit={submit} noValidate>
      <label>
        Customer name
        <input value={customer} onChange={(e) => setCustomer(e.target.value)} placeholder="e.g. Ana Dela Cruz" autoFocus />
        {errors.customer && <span className="error">{errors.customer}</span>}
      </label>

      <fieldset>
        <legend>Items</legend>
        {lines.map((l, i) => (
          <div className="line" key={i}>
            <select value={l.menuId} onChange={(e) => setLine(i, { menuId: e.target.value })} aria-label="Menu item">
              <option value="">Choose an item…</option>
              {menu.map((m) => (
                <option key={m.id} value={m.id} disabled={!m.available && m.id !== l.menuId}>
                  {m.name} — {peso(m.price)}{m.available ? '' : ' (sold out)'}
                </option>
              ))}
            </select>
            <input type="number" min="1" value={l.qty} onChange={(e) => setLine(i, { qty: e.target.value })} aria-label="Quantity" />
            <button type="button" className="icon-btn" onClick={() => removeLine(i)} aria-label="Remove line">✕</button>
          </div>
        ))}
        {errors.items && <span className="error">{errors.items}</span>}
        <button type="button" className="btn btn-sm btn-ghost" onClick={() => setLines((ls) => [...ls, { menuId: '', qty: 1 }])}>+ Add another item</button>
      </fieldset>

      <div className="row">
        <label>
          Status
          <select value={status} onChange={(e) => setStatus(e.target.value)}>
            {ORDER_STATUSES.map((s) => <option key={s}>{s}</option>)}
          </select>
        </label>
        <label>
          Note (optional)
          <input value={note} onChange={(e) => setNote(e.target.value)} placeholder="e.g. oat milk" />

      </div>

  );
}

/**
 * Data service layer.
 *
 * The backend (Express + MongoDB) is not built yet, so this file stores data in
 * the browser's localStorage. Every function is async and has the same shape a
 * real REST call would have. When the backend is ready, replace the bodies with
 * fetch/axios calls, e.g.:
 *
 *   export const getMenuItems = () => fetch(`${API_URL}/menu`).then(r => r.json());
 *
 * Nothing else in the app needs to change.
 */
import { seedMenu, seedOrders } from '../data/seed.js';

const KEYS = { menu: 'mcjoe_menu', orders: 'mcjoe_orders' };

const read = (key, seed) => {
  try {
    const raw = localStorage.getItem(key);
    if (raw) return JSON.parse(raw);
  } catch {
    /* ignore and fall back to seed */
  }
  write(key, seed);
  return seed;
};

const write = (key, value) => {
  try {
    localStorage.setItem(key, JSON.stringify(value));
  } catch {
    /* storage unavailable: data lasts for this session only */
  }
};

const newId = () => Date.now().toString(36) + Math.random().toString(36).slice(2, 6);

/* ---------- Menu items ---------- */
export const getMenuItems = async () => read(KEYS.menu, seedMenu);

export const createMenuItem = async (data) => {
  const list = read(KEYS.menu, seedMenu);
  const item = { ...data, id: newId() };
  write(KEYS.menu, [item, ...list]);
  return item;
};

export const updateMenuItem = async (id, data) => {
  const list = read(KEYS.menu, seedMenu);
  const next = list.map((m) => (m.id === id ? { ...m, ...data, id } : m));
  write(KEYS.menu, next);
  return next.find((m) => m.id === id);
};

export const deleteMenuItem = async (id) => {
  write(KEYS.menu, read(KEYS.menu, seedMenu).filter((m) => m.id !== id));
  return id;
};

/* ---------- Orders ---------- */
export const getOrders = async () => read(KEYS.orders, seedOrders);

export const createOrder = async (data) => {
  const list = read(KEYS.orders, seedOrders);
  const order = { ...data, id: newId(), createdAt: new Date().toISOString() };
  write(KEYS.orders, [order, ...list]);
  return order;
};

export const updateOrder = async (id, data) => {
  const list = read(KEYS.orders, seedOrders);
  const next = list.map((o) => (o.id === id ? { ...o, ...data, id } : o));
  write(KEYS.orders, next);
  return next.find((o) => o.id === id);
};

export const deleteOrder = async (id) => {
  write(KEYS.orders, read(KEYS.orders, seedOrders).filter((o) => o.id !== id));
  return id;
};

import { useCallback, useEffect, useState } from 'react';
import { Routes, Route } from 'react-router-dom';
import Navbar from './components/Navbar.jsx';
import Footer from './components/Footer.jsx';
import Toast from './components/Toast.jsx';
import Home from './pages/Home.jsx';
import Menu from './pages/Menu.jsx';
import ManageMenu from './pages/ManageMenu.jsx';
import Orders from './pages/Orders.jsx';
import NotFound from './pages/NotFound.jsx';
import * as api from './services/api.js';

export default function App() {
  const [menu, setMenu] = useState([]);
  const [orders, setOrders] = useState([]);
  const [loading, setLoading] = useState(true);
  const [toast, setToast] = useState(null);

  const notify = useCallback((message, type = 'ok') => {
    setToast({ message, type, key: Date.now() });
  }, []);

  useEffect(() => {
    Promise.all([api.getMenuItems(), api.getOrders()]).then(([m, o]) => {
      setMenu(m);
      setOrders(o);
      setLoading(false);
    });
  }, []);

  /* ---- menu CRUD ---- */
  const addMenuItem = async (data) => {
    const item = await api.createMenuItem(data);
    setMenu((prev) => [item, ...prev]);
    notify(`“${item.name}” added to the menu`);
  };
  const editMenuItem = async (id, data) => {
    const item = await api.updateMenuItem(id, data);
    setMenu((prev) => prev.map((m) => (m.id === id ? item : m)));
    notify(`“${item.name}” updated`);
  };
  const removeMenuItem = async (id) => {
    const name = menu.find((m) => m.id === id)?.name;
    await api.deleteMenuItem(id);
    setMenu((prev) => prev.filter((m) => m.id !== id));
    notify(`“${name}” deleted`, 'warn');
  };

  /* ---- order CRUD ---- */
  const addOrder = async (data) => {
    const order = await api.createOrder(data);
    setOrders((prev) => [order, ...prev]);
    notify(`Order for ${order.customer} created`);
  };
  const editOrder = async (id, data) => {
    const order = await api.updateOrder(id, data);
    setOrders((prev) => prev.map((o) => (o.id === id ? order : o)));
    notify('Order updated');
  };
  const removeOrder = async (id) => {
    await api.deleteOrder(id);
    setOrders((prev) => prev.filter((o) => o.id !== id));
    notify('Order deleted', 'warn');
  };

  return (
    <div className="app">
      <Navbar />
      <main>
        {loading ? (
          <p className="loading">Brewing…</p>
        ) : (
          <Routes>
            <Route path="/" element={<Home menu={menu} orders={orders} />} />
            <Route path="/menu" element={<Menu menu={menu} />} />
            <Route
              path="/manage-menu"
              element={<ManageMenu menu={menu} onAdd={addMenuItem} onEdit={editMenuItem} onDelete={removeMenuItem} />}
            />
            <Route
              path="/orders"
              element={<Orders orders={orders} menu={menu} onAdd={addOrder} onEdit={editOrder} onDelete={removeOrder} />}
            />
            <Route path="*" element={<NotFound />} />
          </Routes>
        )}
      </main>
      <Footer />
      <Toast toast={toast} onDone={() => setToast(null)} />
    </div>
  );
}
